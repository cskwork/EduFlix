/* ========================================
   Inline EduFlix Engine
   ======================================== */
class Scene {
    constructor(id, element) {
        this.id = id;
        this.element = element;
        this.onEnter = null;
        this.onExit = null;
    }

    show() {
        this.element.classList.add('active');
        if (this.onEnter) this.onEnter();
    }

    hide() {
        this.element.classList.remove('active');
        if (this.onExit) this.onExit();
    }
}

class EduFlixEngine {
    constructor() {
        this.scenes = new Map();
        this.currentSceneId = null;
        this.data = {};
        this.container = document.getElementById('scene-container');
        if (!this.container) {
            this.container = document.createElement('main');
            this.container.id = 'scene-container';
            document.body.appendChild(this.container);
        }
        this.progressEl = null;
    }

    init(gameData) {
        this.data = gameData;
        this.createProgressBar();
        this.createScenes();
        this.start();
    }

    createProgressBar() {
        this.progressEl = document.createElement('div');
        this.progressEl.className = 'progress-bar';
        this.progressEl.innerHTML = '<div class="progress-fill"></div>';
        this.container.appendChild(this.progressEl);
    }

    updateProgressBar(index) {
        const total = 5; // hook, story, core, quiz, wrap
        const fill = this.progressEl.querySelector('.progress-fill');
        if (fill) {
            fill.style.width = `${((index + 1) / total) * 100}%`;
        }
    }

    escapeHtml(value) {
        return String(value).replace(/&/g, '&amp;').replace(/</g, '&lt;').replace(/>/g, '&gt;').replace(/"/g, '&quot;').replace(/'/g, '&#39;');
    }

    createScenes() {
        this.createScene('hook', (scene) => {
            const { question, subText, visual } = this.data.hook;
            scene.innerHTML = `
                <div class="hook-content">
                    <div class="objective-box">🎯 학습 목표: 동물의 환경 적응 특징 알아보기</div>
                    <h1 class="scene-title">${question}</h1>
                    ${subText ? `<p class="scene-text">${subText}</p>` : ''}
                    <div class="visual-area">${visual.content}</div>
                    <button class="btn btn-primary" onclick="Engine.nextScene()">탐험 시작하기</button>
                </div>
            `;
        });

        this.createScene('story', (scene) => {
            const { character, situation } = this.data.story;
            scene.innerHTML = `
                <div class="story-content">
                    <h2 class="scene-title">환경 탐험</h2>
                    <div style="width:80px; height:80px;">${character.image}</div>
                    <p class="scene-text">${situation}</p>
                    <div class="story-stage">
                        <div id="env-display" class="visual-area">
                            <svg viewBox="0 0 100 100"><circle cx="50" cy="50" r="40" fill="#E2E8F0" /></svg>
                        </div>
                        <p id="env-text" class="scene-text">버튼을 눌러 환경을 관찰해 보세요!</p>
                        <div class="env-buttons">
                            <button class="env-btn" id="btn-desert" data-env="desert">사막</button>
                            <button class="env-btn" id="btn-forest" data-env="forest">열대우림</button>
                            <button class="env-btn" id="btn-polar" data-env="polar">극지방</button>
                        </div>
                    </div>
                    <div id="story-feedback" class="feedback-box"></div>
                    <button class="btn btn-primary" id="story-next-btn" disabled onclick="Engine.nextScene()">게임하러 가기</button>
                </div>
            `;
        }, () => { if (this.data.story.onInit) this.data.story.onInit(document.getElementById('scene-quiz')?.parentElement || document.getElementById('scene-story'), this); });

        this.createScene('core', (scene) => {
            scene.innerHTML = `
                <div class="core-content">
                    <h2 class="scene-title">동물 구조 작전</h2>
                    <p class="scene-text">${this.data.interaction.instruction}</p>
                    <div id="core-interactive-area" style="width:100%; display:flex; flex-direction:column; align-items:center;"></div>
                    <div id="core-feedback" class="feedback-box"></div>
                    <button class="btn btn-primary" id="core-next-btn" style="display:none;" onclick="Engine.nextScene()">퀴즈 풀러 가기</button>
                </div>
            `;
        }, () => { if (this.data.interaction.onInit) this.data.interaction.onInit(document.getElementById('core-interactive-area'), this); });

        this.createScene('quiz', (scene) => {
            scene.innerHTML = `
                <div class="quiz-content">
                    <div id="quiz-boss-area" class="visual-area"></div>
                    <h2 class="scene-title" id="quiz-title"></h2>
                    <div id="quiz-question" class="scene-text"></div>
                    <div id="quiz-options" class="quiz-container"></div>
                    <div id="quiz-feedback" class="feedback-box"></div>
                </div>
            `;
        }, () => { this.startQuiz(); });

        this.createScene('wrap', (scene) => {
            scene.innerHTML = `
                <div class="wrap-content">
                    <h1 class="scene-title">환경 적응 대작전 완료!</h1>
                    <div class="visual-area">
                        <img src="/contents/icons/icon-trophy-achievement-cute-20260124.svg" alt="완성을 상징하는 트로피 보상 아이콘" class="character-svg" />
                    </div>
                    <div class="story-stage">
                        <p class="scene-text">오늘 배운 동물들의 환경 적응 비밀을 도감에서 확인해 보세요!</p>
                        <div class="env-buttons">
                            <button class="env-btn" id="polar-book-btn" data-book="polar">북극 도감 보기</button>
                            <button class="env-btn" id="desert-book-btn" data-book="desert">사막 도감 보기</button>
                        </div>
                    </div>
                    <div style="display:flex; gap:15px;">
                        <button class="btn btn-secondary" onclick="location.reload()">다시하기</button>
                        <button class="btn btn-primary" onclick="window.parent.postMessage('close', '*')">완료하기</button>
                    </div>
                </div>
                <div id="dex-overlay" class="dex-overlay hidden" onclick="Engine.closeDex()"></div>
                <div id="dex-popup" class="dex-popup hidden"></div>
            `;
        }, () => { if (this.data.wrap && this.data.wrap.onInit) this.data.wrap.onInit(document.getElementById('scene-wrap'), this); });
    }

    createScene(id, renderFn, onEnterFn) {
        let sceneEl = document.createElement('section');
        sceneEl.id = `scene-${id}`;
        sceneEl.className = `scene scene-${id}`;
        renderFn(sceneEl);
        this.container.appendChild(sceneEl);
        const scene = new Scene(id, sceneEl);
        if (onEnterFn) scene.onEnter = onEnterFn;
        this.scenes.set(id, scene);
    }

    switchScene(sceneId) {
        const order = ['hook', 'story', 'core', 'quiz', 'wrap'];
        const idx = order.indexOf(sceneId);
        if (idx !== -1) this.updateProgressBar(idx);

        if (this.currentSceneId) this.scenes.get(this.currentSceneId).hide();
        if (this.scenes.has(sceneId)) {
            this.currentSceneId = sceneId;
            this.scenes.get(sceneId).show();
        }
    }

    nextScene() {
        const order = ['hook', 'story', 'core', 'quiz', 'wrap'];
        const currentIndex = order.indexOf(this.currentSceneId);
        if (currentIndex < order.length - 1) this.switchScene(order[currentIndex + 1]);
    }

    start() { this.switchScene('hook'); }

    showFeedback(msg, type='neutral', targetId='core-feedback') {
        const el = document.getElementById(targetId);
        if(el) {
            el.innerHTML = msg;
            el.className = `feedback-box feedback-${type}`;
        }
    }

    enableNext(btnId = 'core-next-btn') {
        const btn = document.getElementById(btnId);
        if(btn) {
            btn.style.display = 'inline-flex';
            btn.classList.add('animate-fade-in');
        }
    }

    closeDex() {
        const popup = document.getElementById('dex-popup');
        const overlay = document.getElementById('dex-overlay');
        if(popup) popup.classList.add('hidden');
        if(overlay) overlay.classList.add('hidden');
    }

    startQuiz() {
        if (!this.data.quiz || !this.data.quiz.questions) return;
        
        this.quizState = {
            currentQ: 0,
            questions: this.data.quiz.questions
        };
        this.renderQuiz();
    }

    renderQuiz() {
        const qData = this.quizState.questions[this.quizState.currentQ];
        const bossArea = document.getElementById('quiz-boss-area');
        
        if (this.quizState.currentQ === 0) {
            bossArea.innerHTML = `
                <svg class="boss-character" id="boss-monster" viewBox="0 0 100 100">
                    <path d="M20,50 Q20,10 50,10 Q80,10 80,50 Q80,90 50,90 Q20,90 20,50 Z" fill="#8B5CF6" />
                    <circle cx="35" cy="40" r="5" fill="#fff" /><circle cx="35" cy="40" r="2.5" fill="#000" />
                    <circle cx="65" cy="40" r="5" fill="#fff" /><circle cx="65" cy="40" r="2.5" fill="#000" />
                    <path d="M30,65 Q50,80 70,65" stroke="#000" stroke-width="2" fill="none" />
                </svg>
            `;
        }

        document.getElementById('quiz-title').innerText = `오염 괴물 물리치기 (${this.quizState.currentQ + 1}/${this.quizState.questions.length})`;
        document.getElementById('quiz-question').innerText = qData.prompt;
        
        const optsContainer = document.getElementById('quiz-options');
        optsContainer.innerHTML = '';
        
        qData.choices.forEach((opt, idx) => {
            const btn = document.createElement('button');
            btn.className = 'quiz-option';
            btn.innerText = opt.text;
            btn.onclick = () => this.checkQuiz(idx, btn);
            optsContainer.appendChild(btn);
        });

        document.getElementById('quiz-feedback').innerHTML = '';
    }

    checkQuiz(selectedIdx, btnElement) {
        const qData = this.quizState.questions[this.quizState.currentQ];
        const opts = document.querySelectorAll('.quiz-option');
        opts.forEach(o => o.style.pointerEvents = 'none');

        const fbEl = document.getElementById('quiz-feedback');
        const boss = document.getElementById('boss-monster');

        if (selectedIdx === qData.correctChoiceIndex) {
            btnElement.classList.add('correct');
            fbEl.innerHTML = `<p class="feedback-positive">${qData.correctFeedback}</p>`;
            if(boss) {
                boss.classList.add('boss-hit');
                boss.style.opacity = Math.max(0.2, 1 - (0.33 * (this.quizState.currentQ + 1)));
            }
            
            setTimeout(() => {
                this.quizState.currentQ++;
                if (this.quizState.currentQ < this.quizState.questions.length) {
                    this.renderQuiz();
                } else {
                    this.nextScene();
                }
            }, 1800);
        } else {
            btnElement.classList.add('incorrect');
            fbEl.innerHTML = `<p class="feedback-negative">${qData.incorrectFeedback} <br><small>힌트: ${qData.hint}</small></p>`;
            
            setTimeout(() => {
                opts.forEach(o => {
                    if(!o.classList.contains('correct')) {
                        o.style.pointerEvents = 'auto';
                        o.classList.remove('incorrect');
